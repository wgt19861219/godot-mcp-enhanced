	var info: Dictionary = {
		"size": {"x": vp.get_visible_rect().size.x, "y": vp.get_visible_rect().size.y},
		"content_scale_factor": vp.content_scale_factor,

	var win := get_window()
	if win == null:
		return info
	info["window_size"] = {"x": win.size.x, "y": win.size.y}
	var vsz: Vector2 = vp.get_visible_rect().size
	if vsz.x > 0.0 and vsz.y > 0.0:

**Interfaces:**
- ① `normalizeDraftResPath(raw: string): string`（纯函数）——trim 后以 `res://` 开头则原样；以 `/` 或盘符开头、含 `\` → 抛 `Error('draft_path must be a project-relative path (absolute paths and backslashes are not allowed)')`；剥掉 `res://` 与首部 `./` 后为空 → 抛 `Error('draft_path must name a file under res://')`；任一路径段为 `..` → 抛 `Error('draft_path contains path traversal')`；否则返回 `res://` + 剥前缀后的路径。调用处随后仍过 `sanitizeResPath`（URL 编码的穿越由它迭代解码后拦）。归一值同时用于 `genDraftSaveScript` 与 `payload.persist.path`。错误统一 `opsErrorResult(INVALID_PARAMS, \`draft_path 非法: ${message}\`)`——穿越类 message 含 `path traversal`，其余透传原文。
- ② `genThemeCreateScript(scenePath: string | null, action: 'create' | 'extract', sourceNodePath?: string, savePath?: string): string`——`create` 时 `scenePath` 可为 null，生成脚本不含 `if not _mcp_load_scene(`；`extract` 必须有场景。`index.ts` 仅在 extract 缺 `scene_path` 时报 `scene_path is required for theme_create (extract)`；`scene_path` 描述末尾补 `;theme_create 仅 extract 必填`（33 字节。`ui` 工具的 `perToolTotal` 距 error 线 14400 只余 123B，文案不得加长）。
- ③ `simulate(parsed: AuthoredNode[], viewport?: { w: number; h: number })`（模块内函数）。继承规则：节点满足 `parentPath === '.'`、anchors 为 `(0, 1, 0, 1)`、offsets 全 0，且父尺寸**不可解或非正**——判 `!ps || !(ps.w > 0) || !(ps.h > 0)`，与 `planRebase` 第 ⑤ 步同口径（根节点没有布局行时，`simulate` 会按零锚给它写入真值的 `{w:0,h:0}` 尺寸条目，所以不能只判「有没有」）；此时若 `viewport` 两维为正，以 `viewport` 作父尺寸走既有 float32 公式算边距与尺寸（边距 `{l:0,t:0,r:W,b:H}`，尺寸 `W×H`）。其余分支不变。子节点随后按文档序正常求解，多层全锚链自然传递。`planRebase` 把 `opts?.viewport` 传入；`simulateMargins(parsed)` 调用形态不变。不传 viewport 时结果与改动前逐字节相同。

**既有缺陷（本规则顺带修掉）**：现状在「场景根没有布局行 + 一级子是全锚点零 offset」的场景下传 `viewport` 会**静默错写**——`simulate` 给根写入真值的 0×0 尺寸 → 该子节点边距被算成 `{0,0,0,0}` 进 margins → `planRebase` 第 ⑤ 步因尺寸非正进 viewport 兜底，`margins.has` 对这份退化边距放行 → 以视口求解出全零锚点。实测现状：`BaseLayer` 被写成 anchor 0/0/0/0、offset 全 0，`dry_run=false` 时落盘。Step 1c 第一条用例同时是它的回归锚。
